\section{Penalty Bound for Lexicographic Set Partitioning}

\begin{definition}[Lexicographic Set Partitioning over Finite Route Pool]
Let $\mathcal{P} = \{R_1, \dots, R_m\}$ be the finite pool of feasible routes accumulated during search, $\mathbf{a}_p \in \{0, 1\}^n$ the customer-incidence binary vector of route $R_p$, and $c_p = \TD(R_p)$ the Euclidean tour length of route $R_p$. The lexicographic set partitioning problem seeks:
\begin{align}
\min_{\mathbf{x} \in \{0, 1\}^m} \text{lex} \left( \sum_{p=1}^m x_p, \; \sum_{p=1}^m c_p x_p \right) \quad \text{s.t.} \quad \sum_{p=1}^m a_{ip} x_p = 1, \; \forall i \in \mathcal{N}.
\end{align}
\end{definition}

\begin{proposition}[Adaptive Certified Scalar Penalty Bound on Finite Route Pool]
Let $\mathcal{P}$ be a finite pool of feasible routes, and let $\mathcal{X}_{\text{SP}}(\mathcal{P}) = \{\mathbf{x} \in \{0, 1\}^{|\mathcal{P}|} : \sum_{p=1}^{|\mathcal{P}|} a_{ip} x_p = 1, \; \forall i \in \mathcal{N}\}$ denote the set of valid partitioning solutions. Let $c_{\max}^{\text{route}} = \max_{p \in \mathcal{P}} c_p$ be the longest single route admitted to $\mathcal{P}$, $\bar{c} = \frac{1}{|\mathcal{P}|}\sum_{p \in \mathcal{P}} c_p$ the pool average route distance, and $K_{\max}$ the model's vehicle ceiling, or the incumbent's fleet size when no ceiling is set. Define:
\begin{equation}
C_{\max}^{\mathcal{P}} = \max_{\mathbf{x} \in \mathcal{X}_{\text{SP}}(\mathcal{P})} \sum_{p=1}^{|\mathcal{P}|} c_p x_p \le K_{\max} \cdot c_{\max}^{\text{route}}, \qquad C_{\min}^{\mathcal{P}} = \min_{\mathbf{x} \in \mathcal{X}_{\text{SP}}(\mathcal{P})} \sum_{p=1}^{|\mathcal{P}|} c_p x_p.
\end{equation}
To enforce certified lexicographic fleet priority with guaranteed clearance margin $\epsilon_{\text{margin}} > 0$, the adaptive vehicle penalty coefficient is set to:
\begin{equation}
\lambda_{\text{pen}}^{\text{cert}} = \max\left( \lambda_{\text{pen}}^{\text{grid}} \cdot \bar{c}, \; (1 + \epsilon_{\text{margin}}) \cdot K_{\max} \cdot c_{\max}^{\text{route}} \right),
\label{eq:adaptive_lambda_supp}
\end{equation}
where $\epsilon_{\text{margin}} = 0.20$ guarantees $\ge 20\%$ clearance above the theoretical distance gap. For any penalty $\Lambda_{\text{pen}} \ge \lambda_{\text{pen}}^{\text{cert}}$, any optimal solution to the single-objective penalized formulation:
\begin{equation}
\mathbf{x}^* \in \arg\min_{\mathbf{x} \in \mathcal{X}_{\text{SP}}(\mathcal{P})} \left( \Lambda_{\text{pen}} \sum_{p=1}^{|\mathcal{P}|} x_p + \sum_{p=1}^{|\mathcal{P}|} c_p x_p \right)
\end{equation}
is guaranteed to attain the minimal vehicle fleet count over $\mathcal{X}_{\text{SP}}(\mathcal{P})$, matching the exact lexicographic hierarchy.
\end{proposition}

\begin{proof}
Let $\mathbf{x}_1, \mathbf{x}_2 \in \mathcal{X}_{\text{SP}}(\mathcal{P})$ be two feasible partitions such that $\sum_{p} x_{1,p} < \sum_{p} x_{2,p}$. Since vehicle counts are strictly integer-valued, we have $\sum_{p} x_{2,p} - \sum_{p} x_{1,p} \ge 1$.
The difference in penalized objective values between $\mathbf{x}_2$ and $\mathbf{x}_1$ is:
\begin{align}
F(\mathbf{x}_2) - F(\mathbf{x}_1) &= \Lambda_{\text{pen}} \left(\sum_p x_{2,p} - \sum_p x_{1,p}\right) + \left(\sum_p c_p x_{2,p} - \sum_p c_p x_{1,p}\right) \\
&\ge \Lambda_{\text{pen}} \cdot 1 - \left(\sum_p c_p x_{1,p} - \sum_p c_p x_{2,p}\right) \\
&\ge \Lambda_{\text{pen}} - (C_{\max}^{\mathcal{P}} - C_{\min}^{\mathcal{P}}) \\
&\ge \Lambda_{\text{pen}} - K_{\max} \cdot c_{\max}^{\text{route}}.
\end{align}
Because $\Lambda_{\text{pen}} \ge \lambda_{\text{pen}}^{\text{cert}} \ge (1 + \epsilon_{\text{margin}}) K_{\max} \cdot c_{\max}^{\text{route}} > K_{\max} \cdot c_{\max}^{\text{route}}$, we obtain:
\begin{equation}
F(\mathbf{x}_2) - F(\mathbf{x}_1) \ge \epsilon_{\text{margin}} \cdot K_{\max} \cdot c_{\max}^{\text{route}} > 0,
\end{equation}
meaning $F(\mathbf{x}_1) < F(\mathbf{x}_2)$. Hence no partition with a larger vehicle count can achieve an objective value lower than or equal to a partition with a strictly smaller vehicle count over $\mathcal{P}$.
\end{proof}

\emph{Scope.} The proposition concerns an optimal solution of the model over the columns that enter it. When no vehicle ceiling is set, the bound on $C_{\max}^{\mathcal{P}}$ is applied to partitions with at most $K_{\max}$ routes, which is enough for the conclusion provided such a partition exists in the model. The solver places at most 800 pool columns in one model, stops each query at a time limit, and tries four smaller penalty scales first; those smaller scales need not satisfy the bound. A partition returned at a time limit need not be optimal. The solver uses this penalty grid and does not solve a two-stage formulation. The proposition therefore gives no guarantee about the fleet size reached by the search as a whole. Lexicographic order of the reported solution rests on the separate incumbent update rule stated in the main text.

\section{Bellman Contraction and the Scope of Convergence Claims for the Entropy-Gated DDQN}

\begin{theorem}[Bellman Contraction under Entropy Confidence Gating]
Consider a finite Markov decision process with discount $\gamma < 1$, and let the behavior policy select actions from scores that blend the Q-values with prior and bandit terms via the normalized Shannon entropy weight $w_{\text{conf}}(\mathbf{s}) \in [0, 1]$. The Bellman optimality operator $\mathcal{T}$ is a $\gamma$-contraction in the $L_\infty$-norm with unique fixed point $Q^*$, and it does not depend on the behavior policy, the replay scheme or the target-network update.
\end{theorem}

\begin{proof}
The Bellman optimality operator on action-value functions is defined as:
\begin{align}
(\mathcal{T} Q)(s, a) = R(s, a) + \gamma \mathbb{E}_{s' \sim \mathcal{P}} \left[ \max_{a'} Q(s', a') \right].
\end{align}
In Double DQN, target values are evaluated via decoupled argmax selection:
\begin{align}
y_t = R_t + \gamma Q(s_{t+1}, \arg\max_{a'} Q(s_{t+1}, a'; \theta); \theta^-).
\end{align}
By the standard non-expansion property of $\max$, for any value functions $Q_1, Q_2$:
\begin{align}
\left| \max_{a} Q_1(s, a) - \max_{a} Q_2(s, a) \right| \le \max_a |Q_1(s, a) - Q_2(s, a)| \le \|Q_1 - Q_2\|_\infty,
\end{align}
which directly implies:
\begin{align}
\|\mathcal{T} Q_1 - \mathcal{T} Q_2\|_\infty \le \gamma \|Q_1 - Q_2\|_\infty.
\end{align}
Because the discount factors satisfy $\gamma < 1$ ($0.97$ for the Micro controller and $0.95$ for the Macro controller), $\mathcal{T}$ is a strict contraction mapping on the complete metric space $(\mathbb{R}^{|\mathcal{S}| \times |\mathcal{A}|}, \|\cdot\|_\infty)$.

The entropy gating coefficient $w_{\text{conf}}(\mathbf{s})$ affects only the behavior policy during data collection. After a warmup that samples from the bandit, the implementation explores with probability $\varepsilon$ by sampling from the prior and bandit, and otherwise takes $\arg\max_a Q_{\text{eff}}(\mathbf{s}, a)$. Because the target in Double DQN does not use $Q_{\text{eff}}$, the resulting transitions affect which parts of $\mathcal{T}$ are sampled but not $\mathcal{T}$ itself, so the gate leaves the fixed point of $\mathcal{T}$ unchanged.

\emph{Scope.} The contraction property establishes only that $Q^*$ is well defined for a finite MDP; it is not a convergence result for the solver. Convergence of tabular Q-learning to $Q^*$ additionally requires a Markov state, infinitely frequent visits of every state-action pair and suitable step sizes. None of these conditions is established here: the controllers observe aggregated features that are not a sufficient statistic of the search state (main text, macro controller section), they use nonlinear function approximation with replay and target networks, and each run is finite. We therefore claim no convergence of $Q(\cdot; \theta)$ to $Q^*$, and the learned controllers are evaluated empirically.
\end{proof}

\section{Worst-Case Work of the Implemented Ejection Chain}
\label{sec:supp_gec}

\begin{proposition}[Polynomial Worst-Case Work]
Let $R_{\text{target}}$ be the route selected for elimination, with $m$ customers. Let $K$ be the number of remaining routes and $L$ the maximum route length reached during the procedure. The implemented procedure performs $O(m K^3 L^4)$ elementary operations in the worst case, and $O(m K^2 L^3)$ when Level 3 is not run. It terminates, and any plan it returns is feasible with one vehicle fewer.
\end{proposition}

\begin{proof}
The procedure processes the $m$ customers once each, by decreasing demand. For one customer, finding the best insertion into one route tries at most $L+1$ positions and replays the modified route for each, at $O(L)$ cost, for a total of $I = O(L^2)$. It does not use cached slack queries.

Level 1 evaluates the best insertion into each of $K$ routes, costing $O(K I) = O(K L^2)$. Level 2 considers each of at most $KL$ customers $v$ for ejection. For each, it evaluates $I$ for $u$ in the reduced route and then at most $K$ routes for $v$, costing $O(KL \cdot K I) = O(K^2 L^3)$. Level 3 runs only when at most 30 routes remain. For each of at most $KL$ choices of $v$, it considers at most $KL$ choices of $w$, and for each pair evaluates at most $K$ routes for $w$, costing $O(KL \cdot KL \cdot K I) = O(K^3 L^4)$. Summing over $m$ customers gives the bound.

All loops are finite, and each customer is processed once, so the procedure terminates. The first customer that cannot be placed ends it without a result. A returned plan is constructed from replayed feasible routes, and the solver validates its feasibility and fleet size before returning it.
\end{proof}

The bound is not small, and the procedure has no candidate beam and no internal time check. The solver checks its deadline after the procedure returns, and the supervisor enforces the cutoff independently. Search progress within one call is therefore not interruptible by the solver.

\section{Derivation of Savelsbergh Forward Time Slack Insertion Delay Condition}

\begin{proposition}[Forward Time Slack Insertion Feasibility]
Let a route visit sequence be $(v_0, v_1, \dots, v_m, v_{m+1})$. For each node $v_k$, let $e_k$ and $l_k$ denote the earliest and latest service start times, $s_k$ the service duration, $t_k$ the arrival time, and $B_k = \max(t_k, e_k)$ the service begin time. The forward time slack at node $v_k$ is defined as:
\begin{equation}
F_k = \min_{k \le j \le m+1} \left\{ l_j - B_j + \sum_{p=k+1}^j W_p \right\},
\end{equation}
where $W_p = B_p - (B_{p-1} + s_{p-1} + d_{p-1, p})$ is the waiting time at node $v_p$.

When inserting a new customer $u$ between $v_i$ and $v_{i+1}$, let $\Delta t_u$ denote the arrival delay shifted to $v_{i+1}$, given by:
\begin{align}
\Delta t_u &= B_u + s_u + d_{u, i+1} - (B_i + s_i + d_{i, i+1}) \nonumber \\
&= \max\bigl(0, \; e_u - (B_i + s_i + d_{i, u})\bigr) + d_{i, u} + s_u + d_{u, i+1} - d_{i, i+1}.
\end{align}
Assume the route up to $v_i$ is unchanged and $u$ itself is served by $l_u$. The modified route remains time-window feasible for all downstream nodes $v_{i+1}, \dots, v_{m+1}$ if and only if:
\begin{equation}
\max\bigl(0, \; \Delta t_u - W_{i+1}\bigr) \le F_{i+1}.
\end{equation}
In particular, $\Delta t_u \le F_{i+1}$ is sufficient, and it is also necessary when $W_{i+1} = 0$.
\end{proposition}

\begin{proof}
Let $B_j'$ and $W_j'$ denote the updated service start times and waiting times for downstream nodes $j \ge i+1$ following the insertion of customer $u$.
At node $v_{i+1}$, the arrival time increases by $\Delta t_u$. If $\Delta t_u \le W_{i+1}$, the arrival delay is entirely absorbed by the existing waiting time at $v_{i+1}$, resulting in $B_{i+1}' = B_{i+1}$ and zero propagated delay to subsequent nodes $j > i+1$.

In general, the shift in service start at $v_{i+1}$ is $\Delta B_{i+1} = \max(0, \Delta t_u - W_{i+1})$, and waiting at later nodes absorbs it in turn:
\begin{equation}
\Delta B_j = \max\Bigl(0, \; \Delta B_{i+1} - \sum_{p=i+2}^j W_p \Bigr), \quad j \ge i+1.
\end{equation}
Downstream feasibility requires $B_j + \Delta B_j \le l_j$ for every $j \in \{i+1, \dots, m+1\}$. Since the original route is feasible, $l_j - B_j \ge 0$, so a zero shift always satisfies the condition, and for a positive shift it is equivalent to:
\begin{align}
\Delta B_{i+1} \le l_j - B_j + \sum_{p=i+2}^j W_p.
\end{align}
Taking the minimum over $j$ gives $\Delta B_{i+1} \le F_{i+1}$. With $F$ and $W$ cached, this check costs $O(1)$ per query. Our implementation stores the equivalent latest feasible arrival time at each position, which is computed from the suffix of the route.
\end{proof}

\section{Algorithm Specification}
\label{sec:supp_full_algorithm}

Algorithm~\ref{alg:supp_hybrid_ddqn_alns} follows the implementation used in the evaluation. Stagnation-triggered pool seeding, reheating and other minor rules are in the published code and are omitted here.

\begin{algorithm}[!htbp]
\caption{Tri-Level Hybrid DDQN-ALNS for VRPTW}
\label{alg:supp_hybrid_ddqn_alns}
\footnotesize
\begin{algorithmic}[1]
\Require Instance $\mathcal{I}$, Hyperparameters $\Theta$, Time budget $T_{\text{budget}}$
\Ensure Best validated plan $s^*$
\State $s_{\text{cur}} \leftarrow \text{BuildGreedyInitialSolution}(\mathcal{I}, \text{heatmap})$
\If{the Guided Ejection Search phase is on}
    \State $s_{\text{cur}} \leftarrow \text{GES}(s_{\text{cur}})$ for $10\%$ of the remaining budget
\EndIf
\State $s^* \leftarrow s_{\text{cur}}$; $\mathcal{P} \leftarrow \text{ExtractRoutes}(s_{\text{cur}})$; $T \leftarrow 0.05\,C(s_{\text{cur}}) / \ln 2$, with instance scaling
\State Initialize $Q_{\text{macro}}$, $Q_{\text{micro}}$, $\phi_{\text{LAC}}$ with random weights; initialize bandits, UCB statistics and penalty multiplier $\lambda$
\While{less than $80\%$ of $T_{\text{budget}}$ has elapsed}
    \If{the first phase is active and its end condition holds}
        \State End the first phase; $s_{\text{cur}} \leftarrow s^*$
    \EndIf
    \State $a_k \leftarrow \pi_{\text{macro}}(\mathbf{s}_k^{\text{macro}})$ if stagnation is at least 12 iterations, else \textsf{DEFAULT}
    \State Map $a_6$ to \textsf{DEFAULT} after the first phase; $T \leftarrow T \cdot \text{temp\_boost}(a_k)$
    \For{each iteration of the segment, while time remains}
        \State Select $(d, r)$: bandit during warmup, then $\varepsilon$-greedy over $Q_{\text{eff}}(\mathbf{s}_t^{\text{micro}}, \cdot)$
        \State $s_{\text{cand}} \leftarrow \text{Repair}(\text{Destroy}(s_{\text{cur}}, d), r)$
        \If{$a_k = a_5$, or a route has at most 4 customers and $t \bmod 4 = 0$, or stagnation is a positive multiple of 20}
            \State Replace $s_{\text{cand}}$ by the GEC result if it uses fewer vehicles
        \EndIf
        \If{$a_k$ allows recombination, none ran in this segment, and the plateau and pool conditions hold}
            \State Refine $s_{\text{cand}}$ with Set Partitioning over $\mathcal{P}$ via HiGHS
        \EndIf
        \If{the first phase is active or $a_k \in \{a_5, a_6\}$}
            \State Decide by penalized Simulated Annealing on $C_{\text{pen}}$
        \ElsIf{$s_{\text{cand}}$ is feasible, keeps the fleet, and increases distance}
            \State Decide by $(p_t \ge \tau_{\text{acc}}(t)) \lor (P_{\text{SA}} > U)$; after warmup, $p_t = \phi_{\text{LAC}}(\mathbf{z}_t)$
        \Else
            \State Reject infeasible candidates and disallowed fleet increases; accept others by the rules in the main text
        \EndIf
        \If{$s_{\text{cand}}$ is accepted}
            \State Optionally apply local search under the time budget; add routes to $\mathcal{P}$; $s_{\text{cur}} \leftarrow s_{\text{cand}}$
            \If{$s_{\text{cand}}$ is feasible and $f(s_{\text{cand}}) \prec f(s^*)$}
                \State $s^* \leftarrow s_{\text{cand}}$; submit $s^*$ for certification
            \EndIf
        \EndIf
        \State $T \leftarrow 0.99975 \cdot \text{temp\_decay\_scale}(a_k) \cdot T$
        \State Compute $R_t$; store the Micro transition; update $Q_{\text{micro}}$, UCB statistics and $\phi_{\text{LAC}}$
    \EndFor
    \State Compute $R_{\text{seg}}$; store the Macro transition when the controller acted; update $Q_{\text{macro}}$
\EndWhile
\State Tail phases until $T_{\text{budget}}$: recombination; fixed-fleet polish; dense column generation; route elimination; distance polish; final recombination
\State \Return $s^*$
\end{algorithmic}
\end{algorithm}
